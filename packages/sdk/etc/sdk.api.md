/**
 * The agent's account under a grant: a viem smart account at the owner's account address that sends user operations
 * and signs typed data with the session key, on the client's chain.
 *
 * **Details**
 *
 * Use it with viem's own tooling: `createBundlerClient({ account, client, transport })`, then `sendUserOperation`
 * and `waitForUserOperationReceipt`. The first operation on a chain enables the session (Smart Sessions' ENABLE
 * mode) and deploys the account if the grant carries its deployment and its factory is staked (Kernel); later ones
 * use the session. `signTypedData` gives an ERC-7739 session signature, where a `Policy.signTypedData` allows it, that
 * the account accepts once the session is enabled, when the typed data's verifying contract asks (viem's
 * `verifyTypedData` asks from its own validator contract, which no signing policy names).
 *
 * The session key signs each user operation hash, and each ERC-7739 nested digest, bound to the grant's permission id:
 * `keccak256(abi.encode(hash, permissionId))`, as Smart Sessions verifies it, so a signature can't be replayed under
 * another session of the same key.
 *
 * With `check` (the default), every operation and signature is checked against the session's policies first, with
 * the limits' usage read from the chain, and a violation rejects with `PolicyViolationError` before the session key
 * signs. The check is advisory: the contracts still decide.
 *
 * Each `lane` (0 to 2^16 - 1) is its own nonce sequence, so lanes send in parallel. Send the
 * first operation, which enables the session, on one lane before using others.
 *
 * **Gotchas**
 *
 * Rejects with `GrantError` (`SignerMismatch`) for a signer that isn't the session key, or fewer signers than its
 * threshold, and `ChainNotCovered` for a chain the grant doesn't cover; `AccountError` for a client without a chain or
 * a lane out of range. `signMessage` is not supported: Smart Sessions verifies typed data only.
 *
 * @stability experimental
 * @category constructors
 * @since 2.0.0
 */

/**
 * Key stores for Node (and Bun and Deno for files): an encrypted file per key, and the OS keychain.

 * @stability experimental
 * @since 2.0.0

import type { Kdf, Store } from "../internal/keystore.ts";
/**
 * A store that keeps each key in an encrypted JSON file (Web3 Secret Storage) in a folder, on every curve. The folder
 * is created `0700` and each file `0600`.
 *
 * **When to use**
 *
 * Use with servers and agents that keep keys on disk across restarts.
 *
 * **Example** (Keys in a folder)
 *
 * ```ts
 * import { Keystore } from "@namera-ai/sdk"
 * import { Keystore as NodeKeystore } from "@namera-ai/sdk/node"
 *
 * const store = NodeKeystore.file({ path: "./keys", password: process.env.KEYS_PASSWORD! })
 * const key = await Keystore.create(store, { curve: "secp256k1", label: "agent-1" })
 * ```
 *
 * @stability experimental
 * @category constructors
 * @since 2.0.0
 */
export declare const file: (options: file.Options) => Store;
/**
 * Types of `Keystore.file`.
 *
 * @stability experimental
 * @since 2.0.0
 */
export declare namespace file {
    /**
     * The folder, the password, and the key derivation (`kdf`, scrypt by default).
     *
     * @stability experimental
     * @category utility types
     * @since 2.0.0
     */
    interface Options {
        readonly path: string;
        readonly password: string;
        readonly kdf?: Kdf | undefined;
    }
}
/**
 * A store that keeps keys in the OS keychain (macOS Keychain, Windows Credential Manager, Linux Secret Service), on
 * every curve, through the optional `@napi-rs/keyring` package.
 *
 * **Gotchas**
 *
 * Install `@napi-rs/keyring` to use it. Without it, or without a keychain service (a Linux server without Secret
 * Service), every operation fails with `KeystoreError` (`Unavailable`).
 *
 * **Example** (Keys in the OS keychain)
 *
 * ```ts
 * import { Keystore } from "@namera-ai/sdk"
 * import { Keystore as NodeKeystore } from "@namera-ai/sdk/node"
 *
 * const store = NodeKeystore.osKeychain({ service: "my-agent" })
 * const key = await Keystore.create(store, { curve: "secp256k1" })
 * ```
 *
 * @stability experimental
 * @category constructors
 * @since 2.0.0
